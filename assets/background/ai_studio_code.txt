// Fast vector hash for Intel EUs
vec4 hash4( vec4 p ) {
    p = fract(p * 0.1031);
    p *= p + 33.33;
    p *= p + p;
    return fract(p);
}

float hash( float p ) {
    p = fract(p * .1031);
    p *= p + 33.33;
    p *= p + p;
    return fract(p);
}

float onoise(vec3 pos) {
    vec3 x = pos * 2.0;
    vec3 p = floor(x);
    vec3 f = fract(x);

    f = f * f * (3.0 - 2.0 * f);
    float n = p.x + p.y * 57.0 + 113.0 * p.z;

    vec4 h1 = hash4(vec4(n) + vec4(0.0, 1.0, 57.0, 58.0));
    vec4 h2 = hash4(vec4(n) + vec4(113.0, 114.0, 170.0, 171.0));

    return mix(mix(mix(h1.x, h1.y, f.x),
                   mix(h1.z, h1.w, f.x), f.y),
               mix(mix(h2.x, h2.y, f.x),
                   mix(h2.z, h2.w, f.x), f.y), f.z);
}

float noise2d(vec2 x){
    x /= 512.0;
    return (texture(NoiseSamplerX, x, -1.0).r-0.5)*2.0;
}

float noise(vec3 x){
    x /= vec3(100.0,180.0,100.0)*3.0;

    x.y = fract(x.y)*512.0;
    float iz = floor(x.y);
    float fz = fract(x.y);
    vec2 a_off = vec2(23.0, 29.0)*(iz)/512.0;
    vec2 b_off = vec2(23.0, 29.0)*(iz+1.0)/512.0;
    float a = texture(NoiseSampler, x.xz + a_off, -1.0).r;
    float b = texture(NoiseSampler, x.xz + b_off, -1.0).r;
    return (mix(a,b,fz)-0.5)*2.0;
}

float linearizeDepth(float depth) {
    float z = depth * 2.0 - 1.0;
    return (2.0 * nearPlane * farPlane) / (farPlane + nearPlane - z * (farPlane - nearPlane));
}

// Single-tap fast noise lookup
float tfbm(vec3 x, int octaves, float lacunarity, float gain, float amplitude){
    return amplitude * noise(x);
}

float cfbm(vec3 x, int octaves, float lacunarity, float gain, float amplitude){
    return amplitude * noise(x);
}

float fbm(vec3 x, int octaves, float lacunarity, float gain, float amplitude){
    return tfbm(x, octaves, lacunarity, gain, amplitude);
}

vec3 getRayDir(vec2 screenUV){
    vec2 uv = (screenUV*2.0)-1.0;
    vec4 n = vec4(uv, 0.0, 1.0);
    vec4 f = vec4(uv, 1.0, 1.0);
    n = proj * n;
    f = proj * f;
    n.xyz /= n.w;
    f.xyz /= f.w;
    return normalize((viewmat*f).xyz - (viewmat*n).xyz);
}

mat2 spin(float angle){
    float c = cos(angle);
    float s = sin(angle);
    return mat2(c, -s, s, c);
}

float distanceSqr(vec2 a, vec2 b){
    vec2 c = a-b;
    return dot(c,c);
}

vec2 nearestPoint(vec2 v, vec2 w, vec2 p){
    float l2 = distanceSqr(v, w);
    float t = clamp(dot(p-v, w-v) / l2, 0.0, 1.0);
    return v + t * (w - v);
}

float minimumDistance(vec2 v, vec2 w, vec2 p){
    float l2 = distanceSqr(v, w);
    if(l2 == 0.0) return distance(p, v);
    vec2 projP = nearestPoint(v,w,p);
    return distance(p, projP);
}

#define atan2(y, x) atan(y, x)

float HenyeyGreenstein(float g, float mu){
    float denom = 1.0 + g * (g - 2.0 * mu);
    denom = sqrt(denom * denom * denom);
    float heyey = (1.0 - g * g) / (4.0 * 3.14 * denom);
    return mix(heyey, 1.0, 0.1);
}

CloudReturn getClouds(vec3 position, int octaveReduction){
    float totalCloud = 0.0;
    float totalBGCloud = 0.0;
    float totalRain = 0.0;
    float totalDust = 0.0;
    float upperLevel = 0.0;
    float emit = 0.0;

    vec2 s = vec2(worldTime);

    vec3 noisePos = position + vec3(s.x, -worldTime/2.0, s.y);
    vec3 cloudNoisePos = position + vec3(worldTime*0.5, 0.0, worldTime*0.5);
    vec3 overrideNoisePos = position + vec3(worldTime*0.25, 0.0, worldTime*0.25);
    vec3 noisePos2 = position + vec3(s.x*4.0, -worldTime/2.0, s.y*4.0);
    vec3 cloudNoisePos2 = position + vec3(worldTime*1.5, 0.0, worldTime*1.5);
    vec3 noisePosIT = position + vec3(s.x, worldTime/2.0, s.y);

    float noise1 = -10.0;
    float noise2 = -10.0;
    float noise3 = -10.0;

    float bdensityNoise = min(noise2d(cloudNoisePos.xz/400.0), 1.0);
    float boverrideNoise = clamp(noise2d(overrideNoisePos.xz/200.0)+1.0, 0.0, 2.0)/2.0;
    float bcloudNoise = min(noise2d(noisePos.xz/30.0), 1.0);
    float bbaseNoise = noise2d(noisePos.xz/90.0);
    float bheightNoise = noise2d(noisePos.zx/90.0);

    float bgc = 0.0;
    float bv = clamp(bdensityNoise-(1.0-overcastPerc), 0.0, 1.0);
    bgc += max(sqrt(sqrt(bv)), 0.0);
    float bgCloudBase = mix(mix(0.0, 150.0, clamp((bbaseNoise+1.0)*0.5, 0.0, 1.0)), 0.0, bv*bv*bv)+layer0height;
    float bgCloudHeight = mix(300.0, 850.0, clamp((bheightNoise+1.0)*0.5, 0.0, 1.0));
    bgc *= mix(clamp((bcloudNoise-0.1)+bv, 0.0, 1.0), 1.0, bv);
    bgc = sqrt(bgc)*0.5;
    bgc *= mix(bgCloudHeight/850.0, 1.0, sqrt(bv));

    if(position.y > layer0height && position.y < layer0height+1000.0 && overcastPerc > 0.001){
        float bgClouds = 0.05;
        float detailNoise = noise(noisePos/90.0);
        float densityNoise = min(bdensityNoise+(detailNoise*0.1), 1.0);
        float cloudNoise = min(bcloudNoise+(detailNoise*0.3), 1.0);
        float baseNoise = bbaseNoise+(detailNoise*0.1);
        float heightNoise = bheightNoise+(detailNoise*0.1);

        float v = clamp(densityNoise-(1.0-overcastPerc), 0.0, 1.0);
        bgClouds += max(sqrt(sqrt(v)), 0.0);

        float base = mix(mix(0.0, 150.0, clamp((baseNoise+1.0)*0.5, 0.0, 1.0)), 0.0, v*v*v)+layer0height;
        float height = mix(300.0, 850.0, clamp((heightNoise+1.0)*0.5, 0.0, 1.0));

        bgClouds *= mix(clamp((cloudNoise-0.1)+(v*v), 0.0, 1.0), 1.0, v*v*v);
        bgClouds *= 1.0 + detailNoise*0.2;

        v = clamp((position.y-base)/height, 0.0, 1.0);
        bgClouds -= v*v;
        bgClouds *= 1.0 - clamp((base-position.y)/50.0, 0.0, 1.0);

        bgClouds = sqrt(sqrt(sqrt(bgClouds)))*0.5;

        totalBGCloud = max(totalBGCloud, bgClouds);
        upperLevel = max(upperLevel, bgClouds);
    }

    if(position.y > layerCheight && position.y < layerCheight+2000.0 && overcastPerc > 0.001){
        float bgClouds = 0.0;
        float detailNoise = noise(noisePos2/150.0);
        float densityNoise = min(noise2d(cloudNoisePos2.xz/800.0)+(detailNoise*0.1), 1.0);
        float warpNoiseX = noise2d(noisePos2.zx/400.0);
        float warpNoiseY = noise2d(noisePos2.xz/400.0);
        float cloudNoise = min(noise2d((noisePos2.xz/200.0) + (vec2(warpNoiseX, warpNoiseY)*100.0))+(detailNoise*0.2), 1.0);
        float baseNoise = noise2d(noisePos2.xz/400.0)+(detailNoise*0.1);
        float heightNoise = noise2d(noisePos2.zx/150.0)+(detailNoise*0.1);

        float bandNoise = noise2d(noisePos2.xz/800.0)+(detailNoise*0.1);
        float bandNoise2 = noise2d(noisePos2.zx/400.0)+(detailNoise*0.1);

        float v = clamp(densityNoise-(1.0-overcastPerc), 0.0, 1.0);
        bgClouds += max(sqrt(sqrt(v)), 0.0);

        float base = mix(mix(0.0, 1000.0, clamp((baseNoise+1.0)*0.5, 0.0, 1.0)), 0.0, v*v*v)+layerCheight;
        float height = mix(200.0, 500.0, clamp((heightNoise+1.0)*0.5, 0.0, 1.0));

        bgClouds *= mix(clamp(sqrt(cloudNoise+0.3)+(v*v), 0.0, 1.0), 1.0, v*v*v);
        bgClouds *= 1.0 + detailNoise*0.2;

        v = clamp((position.y-base)/height, 0.0, 1.0);
        bgClouds -= v*v;
        bgClouds *= 1.0 - clamp((base-position.y)/50.0, 0.0, 1.0);

        v = clamp(densityNoise, 0.0, 1.0);
        bgClouds = sqrt(sqrt(bgClouds))*mix(0.35, 0.0, v*v*v*v*v);

        totalBGCloud = max(totalBGCloud, bgClouds);
    }

    int activeStorms = min(stormCount, 8);
    for(int i = 0; i < activeStorms; i++){
        vec3 pos = vec3(stormPositions[i*3], stormPositions[(i*3)+1], stormPositions[(i*3)+2]);
        float stormType = stormTypes[i];
        float width = max(tornadoWidths[i], 15.0);

        vec2 dXZ = position.xz - pos.xz;
        float distSq = dot(dXZ, dXZ);

        float maxBound = stormSize * 16.0 + 500.0;
        if(abs(stormType-2.0) < 0.1) maxBound = max(maxBound, width * 2.0);
        if(distSq > maxBound * maxBound && position.y < layer0height + 1500.0) continue;

        float clouds = 0.0;
        float rainam = 0.0;
        vec2 vel = vec2(stormVelocities[i*2], stormVelocities[(i*2)+1]);
        float stage = stormStages[i];
        float energy = stormEnergies[i];
        float windspeed = tornadoWindspeeds[i];
        float touchdownSpeed = tornadoTouchdownSpeeds[i];
        float tornadoShape = tornadoShapes[i];
        float rainIntensity = rainIntensities[i];
        float stormSpin = stormSpins[i];
        float occlusion = stormOcclusions[i];
        bool visualOnly = (visualOnlys[i] > 0.0);
        bool tornadic = false;

        float smoothStage = stage + (energy/100.0);

        // Fire Whirl
        if(abs(stormType-3.0) < 0.1){
            if(distSq > (width * 3.0 + 200.0) * (width * 3.0 + 200.0)) continue;

            float torPerc = clamp(windspeed/55.0, 0.0, 1.0);
            float fnlTop = pos.y + mix(15.0, 85.0, torPerc);
            float tornadoHeight = pos.y - 1.0;

            float tornado = 1.0;
            float percFnlHeight = clamp((position.y-pos.y)/(fnlTop-pos.y), 0.0, 1.0);
            float percCos = (-cos(percFnlHeight*3.141592)+1.0)*0.5;

            float wid = (width/5.0) + (20.0*percFnlHeight*percFnlHeight*percFnlHeight*torPerc);
            wid = mix(wid, 0.0, (1.0-percFnlHeight)*(1.0-torPerc));
            float th = 1.0 - clamp((position.y-tornadoHeight)/30.0, 0.0, 1.0);
            wid = mix(wid, 0.0, th*th*th);
            float maxWid = width;
            vec3 torPos = pos;

            float ropeMod = 3.0;

            float nx = noise2d(pos.xz/500.0 + vec2(time/200.0))*5.0*ropeMod;
            float nz = noise2d(pos.zx/500.0 + vec2(time/200.0))*5.0*ropeMod;
            vec3 attachmentPoint = vec3(nx, 0.0, nz);

            float xAdd = noise2d(pos.xz/250.0 + vec2((time/200.0)+((position.y*ropeMod)/50.0)))*3.0*ropeMod;
            float zAdd = noise2d(pos.zx/250.0 + vec2((time/200.0)+((position.y*ropeMod)/50.0)))*3.0*ropeMod;

            float a = pow(percFnlHeight, 0.75);
            xAdd *= a;
            zAdd *= a;

            torPos += mix(vec3(0.0), vec3(attachmentPoint.x, 0.0, attachmentPoint.z), percCos);
            torPos += vec3(xAdd, 0.0, zAdd);

            float torDist = distance(torPos.xz, position.xz);
            vec3 localTorPos = position-torPos;

            float widPerc = 1.0 - clamp(torDist/wid, 0.0, 1.0);
            float widMaxPerc = clamp(wid/maxWid, 0.0, 1.0);
            float rotation = -stormSpin*3.0;
            float rotation2 = -stormSpin/1.5;

            mat2 torSpin = spin(rotation+(torDist/50.0));
            mat2 torSpin2 = spin(rotation2+(torDist/150.0));
            vec3 torSpinPos = vec3(torSpin*localTorPos.xz, position.y-(time/2.0));
            vec3 torSpinPos2 = vec3(torSpin2*localTorPos.xz, position.y-(time/2.0));

            float nComp1 = noise(torSpinPos/20.0);
            float nComp2 = noise(torSpinPos2/40.0);

            float torNoise1 = mix(nComp1, nComp2, sqrt(widMaxPerc));

            wid *= mix(0.8 + (torNoise1*0.2), 0.9, clamp(width/1000.0, 0.0, 1.0)*0.9);

            widPerc = 1.0 - clamp(torDist/wid, 0.0, 1.0);

            tornado *= widPerc;
            tornado = tornado * sqrt(tornado) * 4.0;
            tornado *= clamp((position.y-tornadoHeight)/20.0, 0.0, 1.0);
            tornado *= 0.8 + (torNoise1*0.2);

            float heightFadeAt = clamp(windspeed * 2.0, 0.0, 85.0);
            tornado *= sqrt(1.0-clamp(localTorPos.y/heightFadeAt, 0.0, 1.0));
            tornado *= 1.0-((1.0-clamp(localTorPos.y/10.0, 0.0, 1.0))*0.3);

            clouds = max(clouds, tornado);
            totalDust = max(totalDust, tornado * 10.0);
            emit = max(emit, pow(1.0-clamp(localTorPos.y / 50.0, 0.0, 1.0), 1.5) * 0.35 * sqrt(sqrt(tornado)));
        }

        // Hurricane
        if(abs(stormType-2.0) < 0.1 && windspeed > 0.0){
            float hHeight = layer0height+1000.0;
            float heightP = (position.y-layer0height)/(hHeight-layer0height);
            float heightP2 = heightP * heightP;
            float sze = width;
            pos += vec3(heightP2*sze*0.1, 0.0, -heightP2*sze*0.05);
            float dist = distance(position.xz, pos.xz);

            sze *= 1.0+(heightP2*0.4*(1.0+(clamp(dist/sze, 0.0, 1.0)*1.0)));

            float eyeSize = 0.2;
            float eyeCutSize = 0.2 + (heightP*0.2);

            if(dist > sze*1.2 || position.y > hHeight){
                continue;
            }

            if(noise1 < -9.0 && (position.y <= hHeight)){
                noise1 = noise(noisePos/90.0);
                noise2 = noise(noisePosIT/120.0);
                noise3 = noise2d(noisePos.xz/25.0)+(noise1*0.4);
            }

            if(noise1 > -9.0){
                dist *= 1.0+(((noise3+1.0)/2.0)*0.2);
            }

            float intensity = pow(clamp(windspeed/65.0, 0.0, 1.0), 0.85);

            vec2 relPos = pos.xz-position.xz;

            float d = sze/(3.0+(windspeed/12.0));
            float d2 = sze/(1.15+(windspeed/12.0));
            float dE = (sze*0.3)/(1.75+(windspeed/12.0));

            float fac = 1.0+(max((dist-(sze*(eyeSize*2.0)))/sze, 0.0)*2.0);
            d *= fac;
            d2 *= fac;

            float angle = ((atan2(relPos.y, relPos.x)-(dist/d)));
            float angle2 = ((atan2(relPos.y, relPos.x)-(dist/d2)));
            float angleE = ((atan2(relPos.y, relPos.x)-(dist/dE)));

            float weak = 0.0;
            float strong = 0.0;
            float intense = 0.0;

            float staticBands = sin(angle-(PI/2.0));
            staticBands *= pow(clamp(dist/(sze*0.25), 0.0, 1.0), 0.1);
            staticBands *= 1.25*pow(intensity, 0.75);
            if(staticBands < 0.0){
                weak += abs(staticBands);
            }else{
                weak += abs(staticBands) * sqrt(1.0-clamp(dist/(sze*0.65), 0.0, 1.0));
                weak *= clamp((windspeed-70.0)/40.0, 0.0, 1.0);
            }

            float rotatingBands = sin((angle2+radians(worldTime/8.0))*6.0);
            rotatingBands *= pow(clamp(dist/(sze*0.25), 0.0, 1.0), 0.1);
            rotatingBands *= 1.25*pow(intensity, 0.75);
            strong += mix((abs(rotatingBands)*0.3)+0.7, weak, 0.45);
            intense += mix((abs(rotatingBands)*0.2)+0.8, weak, 0.3);
            weak = ((abs(rotatingBands)*0.3)+0.6)*weak;

            float localCloud = 0.0;

            localCloud += mix(mix(weak, strong, clamp((windspeed-40.0)/90.0, 0.0, 1.0)), intense, clamp((windspeed-120.0)/60.0, 0.0, 1.0));

            float eye = sin((angleE+radians(worldTime/4.0))*2.0);
            eye = mix(eye, 1.0, 1.0-clamp(dist/(sze*eyeSize), 0.0, 1.0));
            localCloud = max(sqrt(1.0-clamp(dist/(sze*eyeSize*2.0), 0.0, 1.0))*(abs(eye*0.3)+0.7)*1.4*intensity, localCloud);

            localCloud *= sqrt(1.0-clamp(dist/sze, 0.0, 1.0));
            float eyeCut = clamp(dist/(sze*eyeCutSize), 0.0, 1.0);
            localCloud *= mix(1.0, eyeCut*eyeCut, 0.5+(clamp((windspeed-65.0)/60.0, 0.0, 1.0)*0.5));

            float noiseMain = (1.0+noise1)/2.0;
            localCloud *= mix(0.8 + noiseMain*0.4, 1.0, clamp((windspeed-75.0)/50.0, 0.0, 1.0));
            localCloud *= 0.8 + noiseMain*0.4;

            float eyeBG = mix(1.0, eyeCut*eyeCut, clamp((windspeed-65.0)/60.0, 0.0, 1.0));
            totalBGCloud *= eyeBG;
            bgc *= eyeBG;

            if(localCloud > 0.7){
                float dif = (localCloud-0.7)/3.0;
                localCloud -= dif;
            }

            if(heightP >= 0.0 && heightP <= 1.0){
                float hFactor = (1.4*heightP)-0.4;
                clouds += localCloud-(0.4*(1.0-(hFactor*hFactor)));
                clouds *= sqrt(1.0-heightP);
            }

            if(heightP <= 1.0){
                float rainVal = max(localCloud-0.15, 0.0)*2.0;
                rainVal *= sqrt(sqrt(1.0-clamp(dist/width, 0.0, 1.0)));
                rainVal *= 1.0 + noise2*0.2;
                rainVal *= sqrt(1.0-heightP);
                rainam += rainVal*rainStrength;
            }

            clouds = sqrt(clouds)*0.8;
        }

        // Squall
        if(abs(stormType-1.0) < 0.1){
            if(stage >= 2.99){
                smoothStage = 3.0;
            }

            float subcellular = clamp(smoothStage, 0.0, 1.0);
            float cellular = clamp(smoothStage-0.5, 0.0, 1.0);

            float stormHeight = mix(400.0, 1600.0, cellular);
            float baseHeight = layer0height;

            float v = clamp((position.y-layer0height)/1600.0, 0.0, 1.0);
            v *= v;
            vec2 right = normalize(vel.yx*vec2(1,-1));
            vec2 fwd = normalize(vel.xy);
            vec3 right3 = vec3(right.x, 0.0, right.y);

            float rawDist = distance(position.xz, pos.xz);

            vec3 l = right3*-(stormSize*5.0);
            vec3 r = right3*(stormSize*5.0);

            float rawDistRatio = clamp(rawDist/(stormSize*5.0), 0.0, 1.0);
            vec3 offset = vec3(-fwd.x, 0.0, -fwd.y)*(rawDistRatio*rawDistRatio)*(stormSize*1.5);
            l += offset;
            r += offset;

            l += vec3(2000.0*v, 0.0, -900.0*v);
            r += vec3(2000.0*v, 0.0, -900.0*v);

            l += pos;
            r += pos;

            float dist = minimumDistance(l.xz, r.xz, position.xz);
            float sze = stormSize*10.0;

            if(position.y > layer0height+1000.0){
                sze *= 8.0;
            }

            if(dist > sze){
                continue;
            }

            if(noise1 < -9.0 && (dist < stormSize*1.6 || smoothStage > 1.0) && (position.y < stormHeight*1.5)){
                noise1 = noise(noisePos/90.0);
                noise2 = noise(noisePosIT/120.0);
                noise3 = noise2d(noisePos.xz/25.0)+(noise1*0.4);
            }

            if(noise1 > -9.0){
                baseHeight += noise1*15.0;
            }

            float noiseMain = noise1;
            v = clamp((position.y-layer0height)/1600.0, 0.0, 1.0);
            noiseMain = mix(noiseMain, noise3+(noise1*0.25), cellular*v);

            float size = stormSize*1.5;
            stormHeight *= 1.0 + noiseMain*0.1;

            offset = vec3(fwd.x, 0.0, fwd.y)*stormSize*0.5;
            l += offset;
            r += offset;
            vec2 nearPoint = nearestPoint(l.xz, r.xz, position.xz);
            vec2 facing = position.xz-nearPoint;
            float behind = -dot(facing, fwd);
            behind += noise3*stormSize*0.2;

            if(behind > 0.0){
                float bhRatio = clamp(1.0-((behind-20.0)/stormSize), 0.0, 1.0);
                float bh4 = bhRatio * bhRatio * bhRatio * bhRatio;
                baseHeight *= 1.0 - (bh4*0.45*clamp(smoothStage-1.0, 0.0, 1.0));

                float heightFromBase = position.y-baseHeight;

                if(position.y < layer0height){
                    clouds -= clamp((heightFromBase-30.0)/30.0, 0.0, 1.0)*2.0;
                }

                float baseChange = sqrt(clamp((behind-stormSize)/(stormSize*4.0), 0.0, 1.0))*300.0;
                baseHeight += baseChange;
                stormHeight -= baseChange;
            }

            float heightFromBase = position.y-baseHeight;
            float heightPerc = clamp(heightFromBase/stormHeight, 0.0, 1.0);
            float currentHeight = heightPerc*stormHeight;
            float heightFromTop = stormHeight-currentHeight;

            v = clamp(currentHeight/max(stormHeight, 1600.0), 0.0, 1.0);
            v *= v;
            pos += vec3(2000.0*v, 0.0, -900.0*v);

            size = mix(size, mix(mix(size, size/2.0, cellular), size, sqrt(1.0-clamp(currentHeight/300.0, 0.0, 1.0))), sqrt(sqrt(heightPerc)));

            v = 1.0-clamp(heightFromTop/1600.0, 0.0, 1.0);
            size = mix(size, mix(size, size*8.0, cellular), v*v*v*v*v);
            size = mix(size, size/4.0, clamp(1.0-smoothStage, 0.0, 1.0));

            size = mix(size, size*4.0, clamp(behind/(stormSize/4.0), 0.0, 1.0)*clamp(smoothStage-1.0, 0.0, 1.0));

            size *= 1.0 + noiseMain*0.4;

            float distPerc = clamp(dist/size, 0.0, 1.0);

            if(distPerc >= 0.999){
                continue;
            }

            if(position.y < baseHeight){
                totalBGCloud *= distPerc*distPerc*distPerc;
            }

            float distBased = 1.0-distPerc;
            float cloudField = (noise3-clamp(1.0-(smoothStage-1.0), 0.0, 1.0))*clamp(heightFromTop/65.0, 0.0, 1.0)*sqrt(sqrt(1.0-distPerc));
            cloudField *= clamp(rawDist/mix(stormSize, stormSize*5.0, clamp(smoothStage/1.25, 0.0, 1.0)), 0.0, 1.0);

            clouds += mix(0.0, mix(cloudField, distBased, clamp(smoothStage/2.0, 0.0, 1.0)), step(baseHeight, position.y)*step(position.y, baseHeight+stormHeight));

            clouds *= clamp(heightFromTop/mix(100.0, 300.0, clamp(smoothStage*0.8, 0.0, 1.0)), 0.0, 1.0);

            size = stormSize*0.75*clamp(smoothStage, 0.0, 1.0);
            size = mix(size, size*8.0, clamp(behind/(stormSize/4.0), 0.0, 1.0)*clamp(smoothStage-1.0, 0.0, 1.0));
            size *= 1.0 + noise2*0.4;
            distPerc = clamp(dist/size, 0.0, 1.0);

            clouds = sqrt(clouds)*0.8;

            float rainVal = step(position.y, baseHeight);
            rainVal *= 1.0-pow(distPerc, 1.35);
            rainVal *= 1.0 + noise2*0.2;
            rainVal *= clamp(smoothStage-0.5, 0.0, 1.0);

            if(behind > 0.0){
                rainVal = rainVal * rainVal * sqrt(rainVal);
            }

            behind -= stormSize/3.0;

            if(behind < 0.0){
                rainVal *= 1.0-clamp(abs(behind)/(stormSize/3.0), 0.0, 1.0);
            }

            rainam += rainVal*rainStrength*rainIntensity;
        }

        // Supercell
        if(abs(stormType) < 0.1){
            if(stage >= 2.99){
                tornadic = true;
                smoothStage = 3.0;
            }

            float visualOnlyStep = step(float(visualOnly), 0.5);
            float subceullular = clamp(smoothStage-0.5, 0.0, visualOnlyStep);
            float supercellular = clamp(smoothStage-1.5, 0.0, visualOnlyStep);

            float stormHeight = mix(400.0, 1600.0, subceullular);
            float baseHeight = layer0height;

            float v = clamp((position.y-layer0height)/1600.0, 0.0, 1.0);
            v *= v;
            float dist = distance(position.xz, pos.xz + vec2(2000.0*v, -900.0*v));
            float coreDist = distance(position.xz, pos.xz + vec2(2000.0, -900.0));
            float sze = stormSize*2.0;

            float clearBGVal = clamp(dist/(sze*2.0), 0.0, 1.0);
            float clearBG = mix(1.0, clearBGVal*clearBGVal, clamp(smoothStage, 0.0, 1.0));
            totalBGCloud *= clearBG;
            bgc *= clearBG;

            if(position.y > layer0height+1000.0){
                sze *= 8.0;
            }else if(coreDist <= sze*2.0){
                sze *= 2.5;
            }

            if(min(coreDist, dist) > sze){
                continue;
            }

            if(noise1 < -9.0 && (dist < mix(stormSize*1.6, stormSize*0.85, 1.0-visualOnlyStep) || smoothStage > 1.0) && (position.y < stormHeight*1.5)){
                noise1 = noise(noisePos/90.0);
                noise2 = noise(noisePosIT/120.0);
                noise3 = noise2d(noisePos.xz/25.0)+(noise1*0.4);
            }

            if(noise1 > -9.0){
                baseHeight += noise1*15.0;
            }

            vec3 localPos = position-pos;
            mat2 speen = spin((-time/(20.0*50.0))+(dist/(stormSize*2.0)));
            mat2 speen2 = spin((-time/(20.0*15.0))+(dist/(stormSize/2.0)));

            vec3 spinNoisePos = vec3(speen*localPos.xz, position.y-time);
            vec3 spinNoisePos2 = vec3(speen2*localPos.xz, position.y-time);

            float spinNoise1 = 0.0;
            float spinNoise2 = 0.0;

            float heightFromBase = position.y-baseHeight;

            if(position.y < baseHeight && smoothStage > 1.75 && position.y > pos.y-50.0 && dist < stormSize){
                spinNoise2 = noise(spinNoisePos2/80.0);
            }else if(position.y > baseHeight && heightFromBase < 200.0 && smoothStage > 1.75 && dist < stormSize*2.5){
                spinNoise1 = noise(spinNoisePos/120.0);
            }

            float noiseMain = mix(noise1, mix(spinNoise1, spinNoise2, step(position.y, baseHeight)), clamp((smoothStage-1.75)*3.0, 0.0, 1.0)*clamp(1.0-(heightFromBase/200.0), 0.0, 1.0));

            v = clamp((position.y-layer0height)/1600.0, 0.0, 1.0);
            noiseMain = mix(noiseMain, noise3+(noise1*0.25), supercellular*v);

            float size = stormSize*1.5;
            stormHeight *= 1.0 + noiseMain*0.1;

            float heightPerc = clamp(heightFromBase/stormHeight, 0.0, 1.0);
            float currentHeight = heightPerc*stormHeight;
            float heightFromTop = stormHeight-currentHeight;

            v = clamp(currentHeight/max(stormHeight, 1600.0), 0.0, 1.0);
            v *= v;
            pos += vec3(2000.0*v, 0.0, -900.0*v);
            dist = distance(position.xz, pos.xz);

            size = mix(size, mix(mix(size, size/2.0, supercellular), size, sqrt(1.0-clamp(currentHeight/300.0, 0.0, 1.0))), sqrt(sqrt(heightPerc)));

            v = 1.0-clamp(heightFromTop/1600.0, 0.0, 1.0);
            size = mix(size, mix(size, size*8.0, supercellular), v*v*v*v*v);
            size = mix(size, size/4.0, clamp(1.0-smoothStage, 0.0, 1.0)*visualOnlyStep);
            size = mix(size, size/2.0, 1.0-visualOnlyStep);

            size *= 1.0 + noiseMain*0.4;
            float distPerc = clamp(dist/size, 0.0, 1.0);
            float coreDistPerc = clamp(coreDist/(size*4.0), 0.0, 1.0);

            if(min(distPerc, coreDistPerc) >= 0.999){
                continue;
            }

            float distBased = 1.0-distPerc;
            float cloudField = (noise3-clamp(1.0-smoothStage, 0.0, 1.0))*clamp(heightFromTop/65.0, 0.0, 1.0)*sqrt(sqrt(1.0-distPerc));

            clouds += mix(0.0, mix(cloudField, distBased, clamp(smoothStage*1.2, 0.0, visualOnlyStep)), step(baseHeight, position.y)*step(position.y, baseHeight+stormHeight));

            clouds *= clamp(heightFromTop/mix(100.0, 300.0, clamp(smoothStage*0.8, 0.0, 1.0)), 0.0, 1.0);

            size *= 0.35;
            distPerc = clamp(dist/size, 0.0, 1.0);

            float wallcloudLower = 120.0*sqrt(sqrt(1.0-distPerc))*clamp((smoothStage-2.0)*3.0, 0.0, 1.0);
            float wallcloudNoiseMod = clamp((spinNoise2+1.0)*0.5, 0.0, 1.0);
            wallcloudLower *= mix(0.5, 1.75, wallcloudNoiseMod*wallcloudNoiseMod);
            float wallcloud = mix(0.0, 1.0-distPerc, step(position.y, baseHeight)*step(baseHeight-wallcloudLower, position.y));
            wallcloud *= clamp((smoothStage-2.0)*5.0, 0.0, 1.0);
            clouds += wallcloud;

            float fnlTop = max(baseHeight-wallcloudLower, pos.y+30.0);
            float torPerc = clamp(windspeed/touchdownSpeed, 0.0, 1.0);
            float tornadoHeight = mix(fnlTop, pos.y-50.0, mix(wallcloudNoiseMod * 0.3, 1.0, torPerc));

            if(tornadic && position.y < baseHeight-wallcloudLower && position.y > pos.y-50.0 && dist < max(width*4.5, stormSize/3.0)){
                float tornado = clamp(torPerc/0.15, 0.0, 1.0);
                float percFnlHeight = clamp((position.y-pos.y)/(fnlTop-pos.y), 0.0, 1.0);
                float percCos = (-cos(percFnlHeight*3.141592)+1.0)*0.5;

                float torShape = mix(tornadoShape, 20.0, pow(clamp(width/500.0, 0.0, 1.0), 1.75));

                float wid = (width/2.5) + ((width/2.5)*percFnlHeight*torPerc) + ((stormSize/mix(torShape+2.0, torShape, torPerc)) * percFnlHeight*percFnlHeight*percFnlHeight*percFnlHeight);
                wid = mix(wid, 0.0, (1.0-percFnlHeight)*(1.0-torPerc));
                float th = 1.0 - clamp((position.y-tornadoHeight)/30.0, 0.0, 1.0);
                wid = mix(wid, 0.0, th*th*th);

                wid *= mix((wallcloudNoiseMod*0.25)+0.75, (wallcloudNoiseMod*0.5)+0.5, percFnlHeight);
                float maxWid = (width/4.0) + ((width/4.0)*torPerc) + ((stormSize/8.0) * torPerc);
                vec3 torPos = pos;

                float ropeMod = mix(3.0, 1.0, clamp(width/30.0, 0.0, 1.0));
                ropeMod = mix(ropeMod, 1.0, clamp((windspeed-65.0)/30.0, 0.0, 1.0));
                ropeMod = mix(0.1, ropeMod, clamp((windspeed/touchdownSpeed)*1.35, 0.0, 1.0));

                float nx = noise2d(pos.xz/500.0 + vec2(time/200.0))*40.0*ropeMod;
                float nz = noise2d(pos.zx/500.0 + vec2(time/200.0))*40.0*ropeMod;
                vec3 attachmentPoint = vec3(nx, 0.0, nz);

                float xAdd = noise2d(pos.xz/250.0 + vec2((time/200.0)+((position.y*ropeMod)/50.0)))*20.0*ropeMod;
                float zAdd = noise2d(pos.zx/250.0 + vec2((time/200.0)+((position.y*ropeMod)/50.0)))*20.0*ropeMod;

                float a = pow(percFnlHeight, 0.75);
                xAdd *= a;
                zAdd *= a;

                torPos += mix(vec3(0.0), vec3(attachmentPoint.x, 0.0, attachmentPoint.z), percCos);
                torPos += vec3(xAdd, 0.0, zAdd);

                float torDist = distance(torPos.xz, position.xz);
                vec3 localTorPos = position-torPos;

                float widPerc = 1.0 - clamp(torDist/wid, 0.0, 1.0);
                float widMaxPerc = clamp(wid/maxWid, 0.0, 1.0);
                float rotation = -stormSpin*3.0;
                float rotation2 = -stormSpin/1.5;

                mat2 torSpin = spin(rotation+(torDist/50.0));
                mat2 torSpin2 = spin(rotation2+(torDist/150.0));
                vec3 torSpinPos = vec3(torSpin*localTorPos.xz, position.y-(time/2.0));
                vec3 torSpinPos2 = vec3(torSpin2*localTorPos.xz, position.y-(time/2.0));

                float nComp1 = noise(torSpinPos/20.0);
                float nComp2 = noise(torSpinPos2/40.0);

                float torNoise1 = mix(nComp1, nComp2, sqrt(widMaxPerc));

                wid *= mix(0.8 + (torNoise1*0.2), 0.9, clamp(width/1000.0, 0.0, 1.0)*0.9);

                widPerc = 1.0 - clamp(torDist/wid, 0.0, 1.0);

                tornado *= widPerc;
                tornado = tornado * sqrt(tornado) * 4.0;
                tornado *= clamp((position.y-tornadoHeight)/20.0, 0.0, 1.0);
                tornado *= 0.8 + (torNoise1*0.2);

                float dust = 1.0;
                float dcNoise1 = 0.0;

                if(position.y < pos.y+20.0+(width/7.5)+25.0){
                    dcNoise1 = sqrt((noise(torSpinPos/60.0) * 0.5) + 0.75) * 0.45;
                }

                float dcPerc = clamp((windspeed-45.0)/30.0, 0.0, 1.0);
                float h = 5.0 + (dcNoise1*dcNoise1*((width/7.5)+25.0));
                float dcTop = pos.y+(max(dcPerc, 0.35)*h);

                if(position.y < dcTop){
                    float percDCHeight = clamp((position.y-(pos.y-10.0))/(dcTop-pos.y), 0.0, 1.0);

                    wid = ((width/2.5) + ((width/2.5)*percFnlHeight*torPerc) + (25.0*(width/200.0))) + (15.0*pow(percDCHeight, 1.5)*pow(dcPerc, 0.75));
                    wid *= mix(1.0, 1.25, dcNoise1);
                    wid *= mix(0.6 + (dcNoise1*0.5), 0.85, clamp(width/500.0, 0.0, 1.0)*0.9);
                    wid *= 0.75;
                    widPerc = 1.0 - clamp(torDist/wid, 0.0, 1.0);
                    widPerc = sqrt(sqrt(widPerc));
                    v = clamp(torDist/(wid*0.9), 0.0, 1.0);
                    widPerc *= v*v*v;
                    dust *= widPerc;
                    dust = dust * sqrt(dust) * 0.15;
                    dust *= clamp((dcTop-position.y)/20.0, 0.0, 1.0);
                    dust *= clamp((position.y-(pos.y-20.0))/20.0, 0.0, 1.0);
                    dust *= 0.8 + (dcNoise1*0.2);
                    dust *= dcPerc;
                    dust *= 1.0-clamp((width-50.0)/200.0, 0.0, 1.0);
                    dust = sqrt(dust) * 0.85;
                    tornado = max(tornado, dust);
                    totalDust = max(totalDust, dust);
                }

                clouds = max(clouds, tornado);
            }

            size = stormSize*0.75*clamp(smoothStage, 0.0, 1.0);
            size *= 1.0 + noise2*0.4;
            distPerc = clamp(dist/size, 0.0, 1.0);
            coreDistPerc = clamp(coreDist/(size*4.0), 0.0, 1.0);

            clouds = sqrt(clouds)*0.8;

            float rainVal = step(position.y, baseHeight)*visualOnlyStep;
            rainVal *= 1.0-pow(distPerc, 1.35);
            rainVal *= 1.0 + noise2*0.2;
            rainVal *= clamp(smoothStage, 0.0, 1.0);
            rainam += rainVal*rainStrength*mix(1.0, (occlusion*0.5)+0.5, clamp(smoothStage-2.0, 0.0, 1.0))*rainIntensity;

            rainVal = step(position.y, baseHeight+1300.0)*visualOnlyStep;
            rainVal *= 1.0-pow(coreDistPerc, 1.35);
            rainVal *= 1.0 + noise2*0.2;
            rainVal *= clamp((smoothStage-2.0)*2.0, 0.0, 1.0);
            rainam = max(rainam, rainVal*rainStrength*rainIntensity);
        }

        totalCloud = max(totalCloud, clouds);
        totalRain = max(totalRain, rainam);
    }

    float bgcRain = boverrideNoise*bgc;

    if(position.y <= bgCloudBase && bgcRain > 0.15){
        if(noise2 < -9.0){
            noise2 = noise(noisePosIT/120.0);
        }

        float rainVal = (bgcRain-0.15)*1.65;
        rainVal *= 1.0 + noise2*0.2;
        totalRain = max(totalRain, rainVal*rainStrength*1.35);
    }

    float cdensity = max(max(totalCloud, totalBGCloud), 0.0);
    float eV = sqrt(sqrt(clamp(cdensity, 0.0, 1.0)));

    return CloudReturn(cdensity, max(totalRain, 0.0), max(totalDust, 0.0), max(upperLevel, 0.0), clamp(emit*eV, 0.0, 1.0));
}

vec3 worldPos(vec2 uv, float depth, mat4 invProj){
    vec4 ndc;
    ndc.xy = (uv-0.5)*2.0;
    ndc.z = (depth-0.5)*2.0;
    ndc.w = 1.0;

    vec4 clip = invProj*ndc;
    vec4 view = viewmat*(clip/clip.w);
    return view.xyz;
}

// SMART TORNADO-AWARE DUAL-SAMPLE RENDERER (MAX 2 SAMPLES, ZERO RAYMARCH LOOPS)
Render render(vec2 uv, float maxDepth){
    vec3 ro = pos;
    vec3 rd = getRayDir(uv);
    float light = clamp((sunDir.y+0.1)/0.3, 0.0, 1.0);

    float moonLight = 0.0;
    float moonPhase = 0.0;
    float phase = HenyeyGreenstein(0.3, dot(rd, sunDir));

    if(sunDir.y < 0.1){
        moonPhase = HenyeyGreenstein(0.3, dot(rd, sunDir*-1.0));
        moonLight = clamp((sunDir.y-0.1)/-0.1, 0.0, 1.0)*0.25;
        moonPhase = clamp(moonPhase*moonPhase*moonPhase*4.0, 0.0, 1.0);
    }
    float maxLight = max(light, moonLight);

    // 1. Calculate Overhead Cloud Plane Intersection
    float targetY = (ro.y < layer0height) ? (layer0height + 300.0) : (layerCheight + 400.0);
    float rdy = abs(rd.y) < 0.005 ? (rd.y < 0.0 ? -0.005 : 0.005) : rd.y;
    float tCloud = (targetY - ro.y) / rdy;

    if (tCloud < 0.0 || tCloud > maxDepth) {
        tCloud = min(maxDepth * 0.95, renderDistance);
    } else {
        tCloud = min(tCloud, min(maxDepth, renderDistance * 1.5));
    }

    // 2. Find closest point along view ray to any active tornado/storm center
    float tTor = -1.0;
    float minDistToRay = 1e9;
    int activeStorms = min(stormCount, 8);

    vec2 rdXZ = rd.xz;
    float lenXZ = length(rdXZ);

    if (lenXZ > 0.001) {
        vec2 nRdXZ = rdXZ / lenXZ;
        for (int i = 0; i < activeStorms; i++) {
            vec3 sPos = vec3(stormPositions[i*3], stormPositions[i*3+1], stormPositions[i*3+2]);
            vec2 rel = sPos.xz - ro.xz;
            float projDist = dot(rel, nRdXZ);
            if (projDist > 0.0) {
                float perpDist = length(rel - nRdXZ * projDist);
                float maxR = max(tornadoWidths[i] * 4.0, stormSize * 3.0);
                if (perpDist < maxR && perpDist < minDistToRay) {
                    minDistToRay = perpDist;
                    tTor = projDist / lenXZ;
                }
            }
        }
    }

    // Sort up to 2 samples (front-to-back)
    float samples[2];
    int numSamples = 0;

    if (tTor > 0.0 && tTor < maxDepth && tTor < renderDistance * 1.8) {
        samples[numSamples++] = tTor;
    }

    if (tCloud > 0.0 && tCloud < maxDepth) {
        if (numSamples == 0 || abs(tCloud - samples[0]) > 20.0) {
            samples[numSamples++] = tCloud;
        }
    }

    if (numSamples == 2 && samples[0] > samples[1]) {
        float tmp = samples[0];
        samples[0] = samples[1];
        samples[1] = tmp;
    }

    vec4 res = vec4(0.0);
    float depth = maxDepth;
    float totalLightning = 0.0;
    float totalFire = 0.0;
    float totalDensity = 0.0;

    // Composite front to back
    for (int s = 0; s < numSamples; s++) {
        float dist = samples[s];
        vec3 p = ro + rd * dist;

        // Force sample Y coordinate to storm level if sampling tornado
        if (s == 0 && tTor > 0.0 && dist == tTor) {
            p.y = clamp(p.y, pos.y + 10.0, layer0height + 200.0);
        }

        CloudReturn clouds = getClouds(p, 0);

        float fog = snow + (ash / 150.0);
        float fogE = clamp(fog / 0.1, 0.0, 1.0);
        clouds.cloudDensity = mix(clouds.cloudDensity, 0.0, fogE);
        clouds.rainDensity = mix(clouds.rainDensity, 0.0, fogE);

        float d = clouds.cloudDensity;
        float r = clouds.rainDensity;

        if (d > 0.001 || r > 0.001 || clouds.dustDensity > 0.001) {
            if (dist < depth) depth = dist;

            float sampleDensity = d * 1.5;
            totalDensity += sampleDensity;
            float rPerc = clamp(r, 0.0, 1.0);

            vec3 rainColor = vec3(0.051, 0.341, 0.275);
            vec3 cloudColor = mix(mix(vec3(1.0), rainColor, clamp(rain, 0.0, 1.0)), rainColor, rPerc);
            vec3 dustColor = vec3(0.2, 0.125, 0.071);
            float dustP = clamp(sqrt(sqrt(clouds.dustDensity)), 0.0, 1.0);
            dustColor = mix(dustColor*2.5, dustColor, dustP);

            cloudColor = mix(cloudColor, skyColor*0.5, 0.6);
            cloudColor = mix(cloudColor, dustColor, clamp(sqrt(sqrt(clouds.dustDensity)), 0.0, 1.0));

            if(clouds.dustDensity <= 0.0){
                cloudColor = mix(lightingColor*maxLight, cloudColor, clamp(d + (r*5.0) + snow, 0.0, 1.0));
            }

            cloudColor = mix(cloudColor, mix(vec3(0.05), vec3(0.741, 0.345, 0.0), clamp(fire/2.0, 0.0, 1.0)), clamp(ash/1.5, 0.0, 1.0));
            float alpha = clamp(sampleDensity, 0.0, 1.0);
            vec4 col = vec4(cloudColor * light * alpha, alpha);

            res += col * (1.0 - res.a);

            int activeLightning = min(lightningCount, 4);
            for (int j = 0; j < activeLightning; j++) {
                vec3 lPos = vec3(lightningStrikes[j*3], layer0height, lightningStrikes[(j*3)+2]);
                float bright = lightningBrightness[j];
                float lDist = distance(lPos.xz, p.xz);
                float l = (1.0 - clamp(lDist / 750.0, 0.0, 1.0));
                totalLightning += l * l * bright * d;
            }

            totalFire += clouds.emit;

            if (res.a > 0.95) break;
        }
    }

    float rv = clamp(1.0-rain, 0.0, 1.0);
    float lightEng = clamp(totalDensity, 0.0, 1.0) * maxLight * rv * rv;
    return Render(res, lightEng, totalLightning, totalFire, depth);
}

float getDistFromDepth(float depth, vec2 uv, mat4 invProj) {
    if (depth >= 1.0) return 1e10;
    vec3 viewPos = worldPos(uv, depth, invProj);
    return length(viewPos);
}

void main(){
    vec2 uv = texCoord*downsample;
    if(uv.x > 1.0 || uv.y > 1.0){
        discard;
    }

    vec2 uvOffset = vec2(0.0);

    float maxTotalRenderDist = hasDHDepth > 0.5 ? max(renderDistance, dhRenderDistance) : renderDistance;
    float sceneDistance = maxTotalRenderDist;
    float vanillaDepth = texture(DepthSampler, uv).r;

    if(vanillaDepth < 1.0){
        sceneDistance = getDistFromDepth(vanillaDepth, uv, proj);
    }

    if(hasDHDepth > 0.5){
        float dhDepth = texture(dhDepthTex0, uv).r;
        if(dhDepth < 1.0){
            float dhDistance = getDistFromDepth(dhDepth, uv, dhProjectionInverse);
            sceneDistance = min(sceneDistance, dhDistance);
        }

        if(vanillaDepth >= 1.0 && dhDepth >= 1.0){
            sceneDistance = maxTotalRenderDist*4.0;
        }
    }else{
        if(vanillaDepth >= 1.0){
            sceneDistance = maxTotalRenderDist*4.0;
        }
    }

    sceneDistance = min(sceneDistance, maxTotalRenderDist*4.0);

    Render renderOut = render(uv + uvOffset, sceneDistance);
    vec4 col = mix(mix(mix(renderOut.col, vec4(lightingColor, renderOut.col.a), renderOut.lightEnergy*renderOut.col.a), vec4(1.0, 0.4, 0.0, renderOut.col.a), renderOut.fireEnergy), vec4(vec3(0.969, 0.855, 1.0) * 2.0, renderOut.col.a), renderOut.lightningEnergy);

    fragColor = vec4(pow(col.rgb, vec3(1.25)), col.a);
}
